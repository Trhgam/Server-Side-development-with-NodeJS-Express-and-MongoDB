Chạy JavaScript Ngoại Vi Trình Duyệt

Môi trường runtime đa nền tảng mã nguồn mở, hỗ trợ xây dựng các ứng dụng mạng có khả năng mở rộng với kiến trúc bất đồng bộ hướng sự kiện.

server.js
const http = require('http');

const server = http.createServer((req, res) => {
  res.writeHead(200, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ status: 'running', framework: 'Node.js' }));
});

server.listen(3000, () => {
  console.log('Server active on port 3000');
});
